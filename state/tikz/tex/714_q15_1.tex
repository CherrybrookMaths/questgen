\newcommand{\npanel}[9]{%

  \draw[line width=0.9pt] plot[domain=-4.3:4.3,samples=90,smooth] ({#1+#3*\x},{#2-#4*exp(-0.5*\x*\x)});
  \draw[<->] ({#1-4.45*#3},#2) -- ({#1+4.45*#3},#2);
  \foreach \k in {-4,...,4} \node[font=\scriptsize] at ({#1+#3*\k},#8) {$\k$};
  \node[font=\scriptsize] at (#1,#9) {Distribution $#7$};
}
